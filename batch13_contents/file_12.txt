import type { TravelSearchRequest } from './travelSearchRequest'
import {
  scoreOption,
  type FinalDecisionScore,
  type RecommendationLevel,
  type ScoreableOption,
} from './decisionScoreEngine'

// ── Provider types ─────────────────────────────────────────────────────────

export type SearchProviderType = 'flight' | 'hotel' | 'activity' | 'transportation'

export interface SearchProvider {
  id: string
  name: string
  type: SearchProviderType
  priority: number
  enabled: boolean
}

// ── Provider search result (raw output from a provider adapter) ────────────

export interface ProviderSearchResult {
  providerId: string
  providerName: string
  providerType: SearchProviderType
  externalId: string
  title: string
  description: string
  currency: string
  price: number
  originalPrice: number | null
  durationMinutes: number | null
  stops: number | null
  rating: number | null
  location: string | null
  cancellationPolicy: string | null
  baggageIncluded: boolean | null
  familyFriendly: boolean | null
  rawMetadata: Record<string, unknown>
  retrievedAt: string
}

// ── Normalized travel option ───────────────────────────────────────────────

export type NormalizedOptionType = 'flight' | 'hotel' | 'activity' | 'transportation'

export interface NormalizedTravelOption {
  id: string
  type: NormalizedOptionType
  title: string
  providerIds: string[]
  price: number
  currency: string
  durationMinutes: number | null
  stops: number | null
  rating: number | null
  location: string | null
  baggageIncluded: boolean | null
  familyFriendly: boolean | null
  refundable: boolean | null
  attributes: Record<string, string | number | boolean | null>
  decisionScore: FinalDecisionScore | null
  recommendationLevel: RecommendationLevel | null
  reasons: string[]
}

// ── Search plan ────────────────────────────────────────────────────────────

export interface SearchPlanItem {
  providerType: SearchProviderType
  required: boolean
  reason: string
}

export interface SearchPlan {
  items: SearchPlanItem[]
  providerTypesNeeded: SearchProviderType[]
}

// ── Orchestration result ───────────────────────────────────────────────────

export interface ProviderError {
  providerId: string
  providerName: string
  error: string
}

export interface SearchOrchestrationResult {
  requestId: string
  startedAt: string
  completedAt: string
  providersQueried: number
  providersSucceeded: number
  providersFailed: number
  rawResultsCount: number
  normalizedResultsCount: number
  duplicateResultsRemoved: number
  rankedOptions: NormalizedTravelOption[]
  errors: ProviderError[]
}

// ── Provider adapter function type ─────────────────────────────────────────

export type ProviderAdapter = (
  provider: SearchProvider,
  request: TravelSearchRequest,
) => ProviderSearchResult[]

// ── Search plan creation ───────────────────────────────────────────────────

export function createSearchPlan(req: TravelSearchRequest): SearchPlan {
  const items: SearchPlanItem[] = []

  items.push({
    providerType: 'flight',
    required: true,
    reason: 'الرحلة تتطلب نقل جوي من مدينة المغادرة إلى الوجهة',
  })

  items.push({
    providerType: 'hotel',
    required: req.durationDays > 0,
    reason: req.durationDays > 0
      ? `إقامة مطلوبة لمدة ${req.durationDays} أيام`
      : 'مدة الإقامة غير محددة',
  })

  const hasInterests =
    req.shoppingInterest > 0 ||
    req.natureInterest > 0 ||
    req.cultureInterest > 0 ||
    req.beachInterest > 0 ||
    req.adventureInterest > 0 ||
    req.entertainmentInterest > 0

  items.push({
    providerType: 'activity',
    required: hasInterests,
    reason: hasInterests ? 'اهتمامات نشطة محددة للأنشطة' : 'لا توجد اهتمامات نشطة',
  })

  items.push({
    providerType: 'transportation',
    required: req.durationDays > 2,
    reason: req.durationDays > 2
      ? 'النقل المحلي مطلوب للرحلات الطويلة'
      : 'رحلة قصيرة، النقل المحلي اختياري',
  })

  const providerTypesNeeded = items
    .filter(i => i.required)
    .map(i => i.providerType)

  return { items, providerTypesNeeded }
}

// ── Validation ─────────────────────────────────────────────────────────────

export function validateProviderResult(result: ProviderSearchResult): string[] {
  const errors: string[] = []
  if (!result.providerId) errors.push('missing providerId')
  if (!result.externalId) errors.push('missing externalId')
  if (!result.title) errors.push('missing title')
  if (!result.currency) errors.push('missing currency')
  if (typeof result.price !== 'number' || result.price < 0) errors.push('invalid price')
  if (!result.providerType) errors.push('missing providerType')
  if (!result.retrievedAt) errors.push('missing retrievedAt')
  return errors
}

// ── Currency normalization ─────────────────────────────────────────────────

const CURRENCY_MAP: Record<string, string> = {
  'ر.س': 'SAR',
  'ريال': 'SAR',
  'ريال سعودي': 'SAR',
  'usd': 'USD',
  '$': 'USD',
  'eur': 'EUR',
  '€': 'EUR',
  'gbp': 'GBP',
  '£': 'GBP',
  'aed': 'AED',
  'د.إ': 'AED',
}

function normalizeCurrency(raw: string): string {
  const lower = raw.toLowerCase().trim()
  if (CURRENCY_MAP[lower]) return CURRENCY_MAP[lower]
  if (CURRENCY_MAP[raw]) return CURRENCY_MAP[raw]
  return raw.toUpperCase().slice(0, 3) || 'SAR'
}

// ── Normalization ──────────────────────────────────────────────────────────

let normalizationCounter = 0

function generateNormalizedId(providerId: string, externalId: string): string {
  normalizationCounter += 1
  return `${providerId}-${externalId}-${normalizationCounter}`
}

export function normalizeProviderResult(result: ProviderSearchResult): NormalizedTravelOption {
  const currency = normalizeCurrency(result.currency)

  const refundable = result.cancellationPolicy
    ? /free|gratuit|مجاني|بدون رسوم/i.test(result.cancellationPolicy)
    : null

  const attributes: Record<string, string | number | boolean | null> = {}
  for (const [key, value] of Object.entries(result.rawMetadata)) {
    if (typeof value === 'string' || typeof value === 'number' || typeof value === 'boolean' || value === null) {
      attributes[key] = value
    } else {
      attributes[key] = JSON.stringify(value)
    }
  }

  return {
    id: generateNormalizedId(result.providerId, result.externalId),
    type: result.providerType,
    title: result.title,
    providerIds: [result.providerId],
    price: Math.round(result.price * 100) / 100,
    currency,
    durationMinutes: result.durationMinutes,
    stops: result.stops,
    rating: result.rating !== null ? Math.round(result.rating * 10) / 10 : null,
    location: result.location,
    baggageIncluded: result.baggageIncluded,
    familyFriendly: result.familyFriendly,
    refundable,
    attributes,
    decisionScore: null,
    recommendationLevel: null,
    reasons: [],
  }
}

// ── Fingerprinting for deduplication ───────────────────────────────────────

export function createResultFingerprint(result: NormalizedTravelOption): string {
  const parts: string[] = []

  if (result.type === 'flight') {
    const origin = result.attributes.origin ?? ''
    const destination = result.attributes.destination ?? ''
    const departureTime = result.attributes.departureTime ?? ''
    const airline = result.attributes.airline ?? ''
    const flightNumber = result.attributes.flightNumber ?? ''
    parts.push('flight', String(origin), String(destination), String(departureTime), String(airline), String(flightNumber))
  } else if (result.type === 'hotel') {
    const normalizedTitle = result.title.toLowerCase().trim().replace(/\s+/g, '-')
    const checkIn = result.attributes.checkInDate ?? ''
    const checkOut = result.attributes.checkOutDate ?? ''
    parts.push('hotel', normalizedTitle, result.location ?? '', String(checkIn), String(checkOut))
  } else if (result.type === 'activity') {
    const normalizedTitle = result.title.toLowerCase().trim().replace(/\s+/g, '-')
    parts.push('activity', normalizedTitle, result.location ?? '', String(result.price), result.currency)
  } else {
    const normalizedTitle = result.title.toLowerCase().trim().replace(/\s+/g, '-')
    parts.push('transportation', normalizedTitle, result.location ?? '', String(result.price), result.currency)
  }

  return parts.join('|')
}

// ── Deduplication ──────────────────────────────────────────────────────────

export function deduplicateResults(results: NormalizedTravelOption[]): {
  unique: NormalizedTravelOption[]
  removedCount: number
} {
  const fingerprintMap = new Map<string, NormalizedTravelOption>()

  for (const result of results) {
    const fp = createResultFingerprint(result)
    const existing = fingerprintMap.get(fp)

    if (!existing) {
      fingerprintMap.set(fp, result)
      continue
    }

    const existingCompleteness = countKnownFields(existing)
    const newCompleteness = countKnownFields(result)
    const isCheaper = result.price < existing.price && result.currency === existing.currency

    if (isCheaper || (newCompleteness > existingCompleteness)) {
      const mergedProviders = new Set([...existing.providerIds, ...result.providerIds])
      fingerprintMap.set(fp, {
        ...result,
        providerIds: Array.from(mergedProviders),
      })
    } else {
      const mergedProviders = new Set([...existing.providerIds, ...result.providerIds])
      fingerprintMap.set(fp, {
        ...existing,
        providerIds: Array.from(mergedProviders),
      })
    }
  }

  return {
    unique: Array.from(fingerprintMap.values()),
    removedCount: results.length - fingerprintMap.size,
  }
}

function countKnownFields(opt: NormalizedTravelOption): number {
  let count = 0
  if (opt.price > 0) count++
  if (opt.durationMinutes !== null) count++
  if (opt.stops !== null) count++
  if (opt.rating !== null) count++
  if (opt.location !== null) count++
  if (opt.baggageIncluded !== null) count++
  if (opt.familyFriendly !== null) count++
  if (opt.refundable !== null) count++
  return count
}

// ── Scoring bridge ─────────────────────────────────────────────────────────

function toScoreableOption(opt: NormalizedTravelOption): ScoreableOption {
  return {
    price: opt.price,
    currency: opt.currency,
    cabin: opt.attributes.cabin !== undefined && opt.attributes.cabin !== null
      ? String(opt.attributes.cabin) : undefined,
    directFlight: opt.stops !== null ? opt.stops === 0 : undefined,
    stops: opt.stops ?? undefined,
    durationMinutes: opt.durationMinutes ?? undefined,
    airline: opt.attributes.airline !== undefined && opt.attributes.airline !== null
      ? String(opt.attributes.airline) : undefined,
    hotelStars: opt.attributes.hotelStars !== undefined && opt.attributes.hotelStars !== null
      ? Number(opt.attributes.hotelStars) : undefined,
    hotelRating: opt.rating ?? undefined,
    familyFriendly: opt.familyFriendly ?? undefined,
    breakfastIncluded: opt.attributes.breakfastIncluded !== undefined
      ? Boolean(opt.attributes.breakfastIncluded) : undefined,
    freeCancellation: opt.refundable ?? undefined,
    amenities: opt.attributes.amenities !== undefined && opt.attributes.amenities !== null
      ? String(opt.attributes.amenities).split(',').map(a => a.trim()).filter(Boolean) : undefined,
    area: opt.location ?? undefined,
    destination: opt.attributes.destination !== undefined && opt.attributes.destination !== null
      ? String(opt.attributes.destination) : undefined,
    activityType: opt.attributes.activityType !== undefined && opt.attributes.activityType !== null
      ? String(opt.attributes.activityType) : undefined,
  }
}

export function scoreNormalizedOption(
  req: TravelSearchRequest,
  option: NormalizedTravelOption,
): NormalizedTravelOption {
  const scoreable = toScoreableOption(option)
  const decisionScore = scoreOption(scoreable, req)
  return {
    ...option,
    decisionScore,
    recommendationLevel: decisionScore.recommendation,
    reasons: decisionScore.reasons,
  }
}

// ── Ranking ────────────────────────────────────────────────────────────────

export function rankTravelOptions(options: NormalizedTravelOption[]): NormalizedTravelOption[] {
  return [...options].sort((a, b) => {
    const aScore = a.decisionScore?.weightedAverage ?? 0
    const bScore = b.decisionScore?.weightedAverage ?? 0
    return bScore - aScore
  })
}

// ── Orchestration ──────────────────────────────────────────────────────────

export function orchestrateMockSearch(
  req: TravelSearchRequest,
  providers: SearchProvider[],
): SearchOrchestrationResult {
  const startedAt = new Date().toISOString()
  const requestId = `orch-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`

  const plan = createSearchPlan(req)
  const neededTypes = new Set(plan.providerTypesNeeded)

  const errors: ProviderError[] = []
  const allRawResults: ProviderSearchResult[] = []
  let providersSucceeded = 0
  let providersFailed = 0

  for (const provider of providers) {
    if (!provider.enabled) continue
    if (!neededTypes.has(provider.type)) continue

    const adapter = MOCK_ADAPTERS[provider.id]
    if (!adapter) {
      providersFailed++
      errors.push({
        providerId: provider.id,
        providerName: provider.name,
        error: 'No adapter registered for provider',
      })
      continue
    }

    try {
      const rawResults = adapter(provider, req)
      const validResults: ProviderSearchResult[] = []
      for (const raw of rawResults) {
        const validationErrors = validateProviderResult(raw)
        if (validationErrors.length > 0) {
          errors.push({
            providerId: provider.id,
            providerName: provider.name,
            error: `Validation failed for ${raw.externalId}: ${validationErrors.join(', ')}`,
          })
          continue
        }
        validResults.push(raw)
      }
      allRawResults.push(...validResults)
      providersSucceeded++
    } catch (err) {
      providersFailed++
      errors.push({
        providerId: provider.id,
        providerName: provider.name,
        error: err instanceof Error ? err.message : 'Unknown adapter error',
      })
    }
  }

  const providersQueried = providersSucceeded + providersFailed

  const normalized = allRawResults.map(normalizeProviderResult)
  const { unique, removedCount } = deduplicateResults(normalized)
  const scored = unique.map(opt => scoreNormalizedOption(req, opt))
  const ranked = rankTravelOptions(scored)

  const completedAt = new Date().toISOString()

  return {
    requestId,
    startedAt,
    completedAt,
    providersQueried,
    providersSucceeded,
    providersFailed,
    rawResultsCount: allRawResults.length,
    normalizedResultsCount: scored.length,
    duplicateResultsRemoved: removedCount,
    rankedOptions: ranked,
    errors,
  }
}
