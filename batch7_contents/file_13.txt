import { supabase } from '../supabaseClient'
import type { FavoriteRow } from '../types'

export interface CreateFavoriteInput {
  destination: string
  notes?: string
}

export const favoriteRepository = {
  async create(input: CreateFavoriteInput): Promise<FavoriteRow | null> {
    const { data, error } = await supabase
      .from('favorites')
      .insert(input)
      .select()
      .maybeSingle()
    if (error) throw error
    return data
  },

  async listByUser(): Promise<FavoriteRow[]> {
    const { data, error } = await supabase
      .from('favorites')
      .select('*')
      .order('created_at', { ascending: false })
    if (error) throw error
    return data ?? []
  },

  async delete(id: string): Promise<boolean> {
    const { error } = await supabase.from('favorites').delete().eq('id', id)
    if (error) throw error
    return true
  },

  async deleteByDestination(destination: string): Promise<boolean> {
    const { error } = await supabase
      .from('favorites')
      .delete()
      .eq('destination', destination)
    if (error) throw error
    return true
  },

  async exists(destination: string): Promise<boolean> {
    const { data, error } = await supabase
      .from('favorites')
      .select('id')
      .eq('destination', destination)
      .maybeSingle()
    if (error) throw error
    return data !== null
  },
}
