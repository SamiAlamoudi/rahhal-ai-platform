import type { RahhalOrder } from './checkoutTypes'

export interface InvoiceLine {
  label: string
  amount: number
  type: 'item' | 'tax' | 'fee' | 'discount' | 'total'
}

export interface Invoice {
  invoiceNumber: string
  orderNumber: string
  bookingNumber: string
  customerReference: string
  lines: InvoiceLine[]
  subtotal: number
  taxes: number
  fees: number
  discount: number
  total: number
  currency: string
  issuedAt: string
  status: 'issued' | 'paid' | 'refunded'
}

export function generateInvoice(order: RahhalOrder): Invoice {
  const lines: InvoiceLine[] = order.cart.items.map(item => ({
    label: item.title,
    amount: item.price,
    type: 'item',
  }))

  lines.push({ label: 'VAT (15%)', amount: order.cart.taxes, type: 'tax' })
  if (order.cart.fees > 0) {
    lines.push({ label: 'Service Fee', amount: order.cart.fees, type: 'fee' })
  }
  if (order.cart.discount > 0) {
    lines.push({ label: `Discount (${order.couponCode ?? 'N/A'})`, amount: -order.cart.discount, type: 'discount' })
  }
  lines.push({ label: 'Total', amount: order.cart.total, type: 'total' })

  return {
    invoiceNumber: order.invoiceNumber ?? `INV-${order.orderNumber}`,
    orderNumber: order.orderNumber,
    bookingNumber: order.bookingNumber,
    customerReference: order.customerReference,
    lines,
    subtotal: order.cart.subtotal,
    taxes: order.cart.taxes,
    fees: order.cart.fees,
    discount: order.cart.discount,
    total: order.cart.total,
    currency: order.cart.currency,
    issuedAt: new Date().toISOString(),
    status: order.status === 'refunded' ? 'refunded' : 'paid',
  }
}
